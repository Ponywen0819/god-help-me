\begin{denotation}[3cm]

\item[IENF]{
  表皮內神經纖維 (Intraepidermal Nerve Fiber)
}

\item[IENFD]{
  表皮內神經纖維密度 (Intraepidermal Nerve Fiber Density)
}

\item[SFN]{
  小纖維神經病變 (Small Fiber Neuropathy)
}

\item[AEL]{
  標註擴張連結 (Annotation Expansion Linker)，本論文所提出之方法
}

\item[ROI]{
  感興趣區域 (Region of Interest)
}

\item[CLAHE]{
  對比受限直方圖均衡化 (Contrast Limited Adaptive Histogram Equalization)
}

\item[MST]{
  最小生成樹 (Minimum Spanning Tree)
}

\item[MSF]{
  最小生成森林 (Minimum Spanning Forest)
}

\item[CC]{
  連通元件 (Connected Component)
}

\item[$I$]{
  輸入之組織切片影像 (Input histological image)
}

\item[$I_{g}$]{
  綠色通道影像 (Green channel image)
}

\item[$M$]{
  表皮遮罩 (Epidermis mask)
}

\item[$M_{\text{ROI}}$]{
  垂直膨脹後之 ROI 遮罩 (Vertically dilated ROI mask)
}

\item[$A$]{
  二值化之神經纖維標註影像 (Binary annotation of nerve fibers)
}

\item[$E$]{
  經前處理強化後之灰階影像 (Preprocessed vesselness-enhanced image)
}

\item[$C$]{
  像素成本圖 (Pixel-level cost map)，$C(p) = \exp(1 - \tilde{E}(p)) - 1$
}

\item[$L$]{
  連通元件標記圖 (Connected component label map)
}

\item[$N$]{
  標註中連通元件之總數 (Number of annotation components)
}

\item[$D(p)$]{
  像素 $p$ 從任一種子經最小成本路徑所累積之成本 (Dijkstra distance)
}

\item[$O(p)$]{
  像素 $p$ 所歸屬之元件索引 (Owner component of pixel $p$)
}

\item[$G_{c}$]{
  元件層圖 (Component-level graph)，節點為連通元件、邊為跨元件相遇點
}

\item[$T$]{
  由 $G_{c}$ 經裁剪後求得之最小生成森林
}

\item[$\tau$]{
  相鄰元件間相遇成本之裁剪閾值 (Edge pruning threshold)
}

\item[$k_{d}$]{
  Dijkstra 擴張之每元件停止係數 (Per-component stopping multiplier)
}

\item[$\sigma$]{
  Sato 線狀結構濾波之尺度參數 (Scale parameter of Sato filter)
}

\item[$r_{d}$]{
  骨架化前橋接路徑之膨脹半徑 (Dilation radius of bridging paths)
}

\item[$\ell_{\min}$]{
  一條分段被認定為有效跨越所需之最小區域長度 (px)
}

\end{denotation}
